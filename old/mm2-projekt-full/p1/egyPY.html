import sympy as sp
t = sp.symbols("t")
y = sp.symbols("y", cls=sp.Function)
the_eq = sp.Eq(y(t).diff(t), sp.sin(t) + y(t))
gen_sol = sp.dsolve(the_eq)
print(gen_sol.rhs)

C1*exp(t) - sin(t)/2 - cos(t)/2
