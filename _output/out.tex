\begin{multi}{trigorder}
Legyen $S,C,T=\sin(107^\circ),\cos(107^\circ),\tan(107^\circ)$. Ekkor:
\item* $T<C<S$
\item  $C<T<S$
\item  $C<S<T$
\item  $S<C<T$
\end{multi}
\begin{multi}{circle}
Válassza ki a lentiek közül egy $K\left[2, 5\right]$ középpontú kör egyenletét.
\item* $+13-10y+y^{2}-4x+x^{2}=0$
\item  $+4-10y+y^{2}+4x+x^{2}=0$
\item  $+20+10y+y^{2}-4x+x^{2}=0$
\item  $+13+10y+y^{2}+4x+x^{2}=0$
\end{multi}
\begin{multi}{linearis}

\item* $10$
\item  $11$
\item  $12$
\item  $8$
\end{multi}
