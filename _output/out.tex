\begin{quiz}{gyakorló feladatok}
\begin{multi}{trigorder}
Legyen $S,C,T=\sin(81^\circ),\cos(81^\circ),\tan(81^\circ)$. Ekkor:
\item* $C<S<T$
\item  $S<C<T$
\item  $S<T<C$
\item  $T<S<C$
\end{multi}
\begin{multi}{circle}
Válassza ki a lentiek közül egy $K\left[6, 3\right]$ középpontú kör egyenletét.
\item* $20-6y+y^{2}-12x+x^{2}=0$
\item  $9-6y+y^{2}+12x+x^{2}=0$
\item  $29+6y+y^{2}-12x+x^{2}=0$
\item  $20+6y+y^{2}+12x+x^{2}=0$
\end{multi}
\begin{multi}{linearis}
Egy elsőfokú $f$ függvényről tudjuk, hogy $f(-6)=-1, f(3)=-4$. Mennyi $f(-24)$ ?
\item* $2$
\item  $3$
\item  $4$
\item  $0$
\end{multi}
\begin{multi}{tarsasag}
Egy $ 11 $ tagú társaság tagjai egymás után lépnek be egy ajtón. Panna kedvenc számai: $ K=\left[6, 11\right] $.
Mi a valsége, hogy Panna  $ K $-beli helyen lép be az ajtón?
\item* $ \frac{2}{11} $
\item  $ \frac{9}{11} $
\item  $ \frac{1}{11} $
\item  $ \frac{10}{11} $
\end{multi}
\begin{multi}{logeq}
A $ \log(x-12)+\log(x+12)=2\log(16) $ egyenlet megoldása:
\item* $ x=20 $
\item  $ x=20, -20 $
\item  $ x=20, 16 $
\item  $ x=12, 16 $
\end{multi}
\begin{multi}{parity}
$$ {\begin{array}{cccc}-6x& -3\sin(-2x)& \sin(x)& -\cos(-7x)\\ e^{x}-e^{-x}& \log(|x|+1)& 2\sin(3x)& -\sin(x+\pi)\end{array}} $$
A fentiek közül a páros függvények száma:
\item* $ 2 $
\item  $ 6 $
\item  $ 5 $
\item  $ 4 $
\end{multi}
