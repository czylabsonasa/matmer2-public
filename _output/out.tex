\begin{quiz}{gyakorló feladatok}
\begin{multi}{suruseg}
Egy $X$ valségi változó sűrűségfüggvénye: 
$$ f(x)=
\begin{cases}
\frac{A}{x^2} & \text{ha } x>2 \\
0             & \text{ máskor } 
\end{cases}
 $$ alakú valamely $A$ számra. Ekkor $A=$
\item* $2$
\item $3$
\item $4$
\item $5$
\end{multi}
\begin{multi}{cedula}
Egy dobozban $11$ cédula van, rajtuk a $11, 11, 13, 32, 12, 32, 13, 32, 22, 21, 23$ számok. Egy véletlenszerűen 
kihúzott cédula esetén jelentse $X$ az első, $Y$ a második számjegyét.
Ekkor $X$ és $Y$
\item* nem függetlenek és $cov(X,Y)=0$
\item függetlenek és $cov(X,Y)=0$
\item nem függetlenek és $cov(X,Y)=1$
\item függetlenek és $cov(X,Y)=-1$
\end{multi}
\begin{multi}{bot}
Egy $ 4 $ méter hosszú botot  egy csapással véletlenszerűen kettétörünk. Legyen $ Y $ 
a keletkezet részek hosszának minimuma. Mennyi $ P(\frac{9}{25}<Y<\frac{1}{2}) $?
\item* $\frac{7}{100}$
\item $\frac{1}{4}$
\item $\frac{9}{50}$
\item $\frac{7}{200}$
\end{multi}
\begin{multi}{kozep}
A $ (0,5) $ intervallumon választunk egy pontot. Legyen $ Y $ a pontunk távolsága az intervallum közepétől.
$ P(\frac{1}{5}<Y<\frac{3}{5}) $?
\item* $\frac{4}{25}$
\item $\frac{6}{25}$
\item $\frac{2}{25}$
\item $\frac{2}{25}$
\end{multi}
\end{quiz}
